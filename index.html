#include #include #include using namespace std; class TextEditor { private: string text; stack undoStack; // Stack storing previous states of text public: TextEditor() { text = ""; } // Append text and save the current state before making changes void typeText(const string& newText) { undoStack.push(text); // Push state onto stack (LIFO) text += newText; cout << "[Typed]: \"" << newText << "\" | Current Text: \"" << text << "\"\n"; } // Revert to previous state void undo() { if (undoStack.empty()) { cout << "[Undo Failed]: No actions left to undo!\n"; return; } // Retrieve top element (most recent state) and pop it text = undoStack.top(); undoStack.pop(); cout << "[Undo Performed] | Restored Text: \"" << text << "\"\n"; } void display() const { cout << "--> Current Document Content: \"" << text << "\"\n\n"; } }; int main() { TextEditor editor; cout << "--- C++ STACK DEMONSTRATION: TEXT EDITOR UNDO SYSTEM ---\n\n"; editor.typeText("Data "); editor.typeText("Structures "); editor.typeText("Stack "); editor.display(); cout << "--- Pressing Undo --- \n"; editor.undo(); editor.display(); cout << "--- Pressing Undo Again --- \n"; editor.undo(); editor.display(); cout << "--- Pressing Undo Again --- \n"; editor.undo(); editor.display(); cout << "--- Attempting Undo on Empty Stack --- \n"; editor.undo(); return 0; }
